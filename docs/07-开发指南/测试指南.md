## 三、单元测试

### 3.1 数据验证测试

**tests/test_validation.py**：
```python
"""测试数据验证功能。"""
import pytest
from custom_components.anti_loss_tag.validation import (
    validate_mac_address,
    validate_poll_interval,
)


class TestValidateMacAddress:
    """测试MAC地址验证。"""
    
    def test_valid_mac_address(self):
        """测试有效的MAC地址。"""
        valid_macs = [
            "AA:BB:CC:DD:EE:FF",
            "aa:bb:cc:dd:ee:ff",
            "00:11:22:33:44:55",
        ]
        for mac in valid_macs:
            assert validate_mac_address(mac) is True
    
    def test_invalid_mac_address(self):
        """测试无效的MAC地址。"""
        invalid_macs = [
            "AA-BB-CC-DD-EE-FF",  # 错误分隔符
            "AABBCCDDEEFF",      # 缺少分隔符
            "AA:BB:CC:DD:EE",    # 太短
            "AA:BB:CC:DD:EE:FF:GG",  # 太长
            "",                  # 空字符串
        ]
        for mac in invalid_macs:
            assert validate_mac_address(mac) is False


class TestValidatePollInterval:
    """测试轮询间隔验证。"""
    
    def test_valid_poll_interval(self):
        """测试有效的轮询间隔。"""
        valid_intervals = [5, 10, 60, 360, 10080]
        for interval in valid_intervals:
            assert validate_poll_interval(interval) is True
    
    def test_invalid_poll_interval(self):
        """测试无效的轮询间隔。"""
        invalid_intervals = [
            -1,    # 负数
            0,     # 零
            4,     # 小于最小值
            10081, # 大于最大值
        ]
        for interval in invalid_intervals:
            assert validate_poll_interval(interval) is False
```

```python
"""测试常量定义。"""
import pytest
from custom_components.anti_loss_tag import const


class TestConstants:
    """测试常量。"""
    
    def test_domain(self):
        """测试DOMAIN常量。"""
        assert const.DOMAIN == "anti_loss_tag"
    
    def test_uuid_lowercase(self):
        """测试所有UUID都是小写。"""
        uuids = [
            const.UUID_SERVICE_FFE0,
            const.UUID_NOTIFY_FFE1,
            const.UUID_POLICY_FFE2,
            const.UUID_ALERT_2A06,
            const.UUID_BATTERY_2A19,
        ]
        for uuid in uuids:
            assert uuid.islower(), f"{uuid} 不是全小写"
            assert uuid.startswith("0000"), f"{uuid} 缺少0000前缀"
            assert uuid.endswith("00805f9b34fb"), f"{uuid} 后缀错误"
    
    def test_default_values(self):
        """测试默认值。"""
        assert const.DEFAULT_POLL_INTERVAL == 360
        assert const.DEFAULT_TIMEOUT == 20
        assert const.MAX_RETRIES == 3
```

### 3.3 设备类测试

**tests/test_device.py**：
```python
"""测试BLEDevice类。"""
import pytest
from unittest.mock import AsyncMock, MagicMock, patch
from custom_components.anti_loss_tag.device import BLEDevice


class TestBLEDevice:
    """测试BLEDevice类。"""
    
    @pytest.mark.asyncio
    async def test_connect_success(self, ble_device, mock_bleak_client):
        """测试连接成功。"""
        mock_bleak_client.is_connected.return_value = False
        mock_bleak_client.connect = AsyncMock()
        
        result = await ble_device.connect()
        
        assert result is True
        mock_bleak_client.connect.assert_called_once()
    
    @pytest.mark.asyncio
    async def test_connect_failure(self, ble_device, mock_bleak_client):
        """测试连接失败。"""
        from bleak.exc import BleakError
        
        mock_bleak_client.connect = AsyncMock(side_effect=BleakError("连接失败"))
        
        result = await ble_device.connect()
        
        assert result is False
        assert ble_device.last_error == "连接失败"
    
    @pytest.mark.asyncio
    async def test_read_battery(self, ble_device, mock_bleak_client):
        """测试读取电量。"""
        mock_bleak_client.read_gatt_char = AsyncMock(return_value=bytes([85]))
        
        battery = await ble_device.read_battery()
        
        assert battery == 85
        mock_bleak_client.read_gatt_char.assert_called_once()
    
    @pytest.mark.asyncio
    async def test_read_battery_failure(self, ble_device, mock_bleak_client):
        """测试读取电量失败。"""
        from bleak.exc import BleakError
        
        mock_bleak_client.read_gatt_char = AsyncMock(side_effect=BleakError("读取失败"))
        
        battery = await ble_device.read_battery()
        
        assert battery == -1
```

## 四、集成测试

### 4.1 连接测试

**tests/test_integration/test_connection.py**：
```python
"""测试连接功能。"""
import pytest
from custom_components.anti_loss_tag import const
from custom_components.anti_loss_tag.device import BLEDevice


@pytest.mark.integration
class TestConnection:
    """连接集成测试。"""
    
    @pytest.mark.asyncio
    async def test_full_connection_flow(self, hass, config_entry):
        """测试完整连接流程。"""
        device = BLEDevice(
            mac_address=config_entry.data["mac_address"],
            name=config_entry.data["name"],
            hass=hass,
        )
        
        # 1. 连接
        assert await device.connect() is True
        
        # 2. 检查连接状态
        assert device.is_connected is True
        
        # 3. 读取电量
        battery = await device.read_battery()
        assert 0 <= battery <= 100
        
        # 4. 断开连接
        await device.disconnect()
        assert device.is_connected is False
```

### 5.1 定义标记

**pytest.ini**：
```ini

    unit: 单元测试（快速）
```

```python

    """集成测试。"""

@pytest.mark.integration

    """慢速测试。"""
    pass
```

## 六、Mock最佳实践

### 6.1 Mock异步函数

```python
# ✅ 正确
mock_client = MagicMock(spec=BleakClient)
mock_client.connect = AsyncMock()

await mock_client.connect()

# ❌ 错误
mock_client = MagicMock()
mock_client.connect = MagicMock()  # 不是异步
```

---

### 6.2 Mock返回值

```python
# ✅ 正确
mock_client.read_gatt_char = AsyncMock(return_value=bytes([85]))

data = await mock_client.read_gatt_char(UUID_BATTERY)
assert data == bytes([85])

# ❌ 错误
mock_client.read_gatt_char = MagicMock(return_value=bytes([85]))  # 不是异步
```

## 十、常见问题

### Q1: 异步测试失败

**错误**：
```
RuntimeError: Event loop is closed
```

**解决**：
```python
# 确保使用 pytest-asyncio
# conftest.py 中设置
pytest_plugins = ('pytest_asyncio',)

# 测试函数标记为 async
@pytest.mark.asyncio
async def test_async_function():
    pass
```

---

### Q2: Mock对象没有返回预期值

**解决**：
```python
# 检查Mock配置
mock_client.read_gatt_char = AsyncMock(return_value=bytes([85]))

# 验证调用
mock_client.read_gatt_char.assert_called_once_with(UUID_BATTERY)

# 查看调用历史
print(mock_client.read_gatt_char.call_args_list)
```
